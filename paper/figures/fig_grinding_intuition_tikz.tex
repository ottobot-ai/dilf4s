% Dormant branch attack intuition — Fig. 9(a) TikZ replacement
% Requires: \usepackage{tikz}
%           \usetikzlibrary{arrows.meta, shapes.geometric, positioning}
%
% Concept: honest chain produces blocks at regular intervals (w≈1 each);
%          adversary's branch sits dormant then extends after a long gap,
%          collecting high ramp weight "for free".

\begin{tikzpicture}[
  hblock/.style={draw=blue!60!black, fill=blue!50!cyan!40,
                 minimum size=7pt, inner sep=0pt},
  ablock/.style={draw=red!70!black, fill=red!50!orange!40,
                 minimum size=8pt, inner sep=0pt, shape=diamond},
  lbl/.style={font=\scriptsize},
  >=Stealth,
  scale=0.88
]

%% ── Honest chain (top row) ──────────────────────────────────────────────────
\draw[->, gray!60, thick] (-0.2, 1.5) -- (12.5, 1.5)
      node[right, gray!70, font=\tiny] {slots};
\node[left, font=\small\bfseries, blue!70!black] at (-0.2, 1.5) {Honest};

% Blocks at regular spacing
\foreach \x in {1, 2.6, 4.2, 5.8, 7.4, 9.0} {
  \node[hblock] at (\x, 1.5) {};
  \node[above, lbl, blue!70!black] at (\x, 1.82) {$w\!\approx\!1$};
}

%% ── Adversary chain (bottom row) ────────────────────────────────────────────
\draw[->, gray!60, thick, dashed] (-0.2, 0) -- (12.5, 0)
      node[right, gray!70, font=\tiny] {slots};
\node[left, font=\small\bfseries, red!70!black] at (-0.2, 0) {Adversary};

% First block (same start)
\node[ablock] (A1) at (1, 0) {};
\node[above, lbl, red!70!black] at (1, 0.35) {$w\!=\!0.4$};

% Long dormant gap, then second block
\node[ablock] (A2) at (10.5, 0) {};
\node[above, font=\small\bfseries, red!70!black] at (10.5, 0.35)
      {$w\!=\!3.8$ \textbf{(!)}};

%% ── Dormant gap annotation ──────────────────────────────────────────────────
\draw[<->, orange!80!black, thick] (1.2, -0.75) -- (10.3, -0.75);
\node[below, font=\small, orange!80!black, align=center] at (5.75, -0.92)
      {Dormant: 20 slots \quad$\Longrightarrow$\quad
       $w = \dfrac{20}{5.2} \approx 3.8$\;\textbf{free!}};

%% ── Callout: earns high weight by waiting ───────────────────────────────────
\draw[->, red!70!black, thick] (10.5, 0.6) -- (10.5, 1.05);
\node[above, font=\footnotesize\bfseries, red!70!black, align=center]
      at (10.5, 1.1) {Earns high weight\\[-2pt]just by waiting};

\end{tikzpicture}
